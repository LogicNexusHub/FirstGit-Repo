# FirstGit-Repo
practice Git commands
git clone <URL>          # Copy/download a GitHub repository to your local machine
git status               # Check the status of files (new, modified, deleted)
git add <file-name>      # Add a specific file to the staging area
git add .                # Add all changed files to the staging area
git commit -m "message"  # Save staged changes to your local repository with a message
git push origin main     # Upload local commits to the main branch on GitHub
git pull origin main     # Download and merge latest changes from GitHub
git branch               # Show all local branches
git checkout <branch>    # Switch to another branch
git checkout -b <branch> # Create and switch to a new branch
git merge <branch>       # Merge a branch into the current branch
git log                  # View commit history
git diff                 # View changes before committing
